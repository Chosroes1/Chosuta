cmake_minimum_required(VERSION 3.24)
project(Chosuta VERSION 0.10.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(WIN32)
 add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
 add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Concurrent Multimedia)
find_package(ICU 67 REQUIRED COMPONENTS i18n uc data)
add_library(chosuta_core
 src/importers/source.cpp src/importers/ust.cpp src/importers/vsqx.cpp
 src/core/model.cpp src/core/source_time.cpp src/core/evaluation.cpp src/core/performance.cpp src/core/performance_io.cpp src/core/performance_rules.cpp src/core/performance_analysis.cpp src/core/appearance.cpp src/core/intervals.cpp src/core/timing.cpp src/core/subtitles.cpp src/core/executable.cpp src/importers/svp.cpp src/pronunciation/pronunciation.cpp src/pronunciation/consonants.cpp src/pronunciation/dictionary.cpp
 src/rules/generator.cpp src/project/storage.cpp src/project/validation.cpp src/render/render.cpp src/render/audio.cpp src/render/waveform.cpp resources/data.qrc)
target_include_directories(chosuta_core PUBLIC src)
target_link_libraries(chosuta_core PUBLIC Qt6::Core Qt6::Gui PRIVATE ICU::i18n ICU::uc ICU::data)
target_compile_definitions(chosuta_core PUBLIC CHOSUTA_VERSION="${PROJECT_VERSION}")
if(MSVC)
 target_compile_options(chosuta_core PRIVATE /W4 /utf-8)
else()
 target_compile_options(chosuta_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(chosuta_ui src/ui/text.cpp src/ui/message_text.cpp src/ui/widget_language.cpp src/ui/window.cpp src/ui/window.h src/ui/appearance.cpp src/ui/performance.cpp src/ui/waveform.cpp src/ui/timeline.cpp src/ui/timeline.h src/ui/preferences.cpp src/ui/settings.cpp src/ui/preview.cpp src/ui/preview.h src/ui/subtitles.cpp)
target_link_libraries(chosuta_ui PUBLIC chosuta_core Qt6::Widgets Qt6::Concurrent Qt6::Multimedia)
qt_add_executable(chosuta src/ui/main.cpp resources/icons.qrc)
set_target_properties(chosuta PROPERTIES WIN32_EXECUTABLE TRUE)
if(WIN32)
 enable_language(RC)
 target_sources(chosuta PRIVATE resources/icons/chosuta.rc)
 target_include_directories(chosuta PRIVATE "$<$<COMPILE_LANGUAGE:RC>:${CMAKE_CURRENT_SOURCE_DIR}/resources/icons>")
 set_property(SOURCE resources/icons/chosuta.rc PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/resources/icons/chosuta.ico")
endif()
target_link_libraries(chosuta PRIVATE chosuta_ui)
add_executable(chosuta-cli src/cli/main.cpp)
target_link_libraries(chosuta-cli PRIVATE chosuta_core)
configure_file(cmake/build-info.json.in build-info.json @ONLY)
option(BUILD_TESTING "Build developer tests" OFF)
include(CTest)
if(BUILD_TESTING)
 if(UNIX)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
   add_test(NAME packaging-preflight COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/packaging_test.py)
  endif()
 endif()
 find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
 add_executable(chosuta-tests tests/core_test.cpp)
 add_dependencies(chosuta-tests chosuta-cli)
 target_link_libraries(chosuta-tests PRIVATE chosuta_core Qt6::Test)
 target_compile_definitions(chosuta-tests PRIVATE CHOSUTA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/private_samples.inc")
  target_compile_definitions(chosuta-tests PRIVATE CHOSUTA_PRIVATE_SAMPLES)
 endif()
 add_test(NAME core COMMAND chosuta-tests)
 add_executable(chosuta-ui-tests tests/ui_test.cpp)
 target_link_libraries(chosuta-ui-tests PRIVATE chosuta_ui Qt6::Test)
 target_compile_definitions(chosuta-ui-tests PRIVATE CHOSUTA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 add_test(NAME ui-workflow COMMAND chosuta-ui-tests)
 set_tests_properties(ui-workflow PROPERTIES TIMEOUT 30)
 add_test(NAME gui-smoke COMMAND chosuta --smoke-test)
 set_tests_properties(gui-smoke PROPERTIES TIMEOUT 15)
 set_tests_properties(core ui-workflow gui-smoke PROPERTIES
  ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
include(GNUInstallDirs)
# Windows portable deployments launch directly from their root. Unix packages
# continue to use GNUInstallDirs; media tools stay relative to the executable.
if(WIN32)
 set(CHOSUTA_RUNTIME_DESTINATION .)
else()
 set(CHOSUTA_RUNTIME_DESTINATION "${CMAKE_INSTALL_BINDIR}")
endif()
install(TARGETS chosuta chosuta-cli RUNTIME DESTINATION "${CHOSUTA_RUNTIME_DESTINATION}")
install(FILES LICENSE THIRD_PARTY_NOTICES.md README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/chosuta)
foreach(readme_translation README.en.md README.ja.md)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${readme_translation}")
  install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/${readme_translation}" DESTINATION ${CMAKE_INSTALL_DATADIR}/chosuta)
 endif()
endforeach()
install(FILES docs/building.zh-CN.md docs/user-guide.zh-CN.md docs/windows.zh-CN.md docs/packaging.zh-CN.md docs/license-review.zh-CN.md docs/publication.zh-CN.md DESTINATION ${CMAKE_INSTALL_DATADIR}/chosuta/docs)
install(DIRECTORY resources/licenses DESTINATION ${CMAKE_INSTALL_DATADIR}/chosuta)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 install(FILES packaging/linux/chosuta.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
 install(FILES resources/icons/chosuta.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/512x512/apps)
 install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/chosuta)
endif()
set(CPACK_PACKAGE_NAME Chosuta)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_VENDOR "Chosuta contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Score-driven lip animation editor")
set(CPACK_PACKAGE_DESCRIPTION "Import SVP, UST and VSQX scores, edit score-driven mouth shapes and export videos with optional FFmpeg. Optional waveforms provide explicitly requested, bounded and reversible timing estimates; audio never selects mouth shapes.")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_GENERATOR ZIP)
configure_file(cmake/CPackOptions.cmake.in CPackOptions.cmake @ONLY)
set(CPACK_PROJECT_CONFIG_FILE "${CMAKE_CURRENT_BINARY_DIR}/CPackOptions.cmake")
include(CPack)
